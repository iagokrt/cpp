float CalculaDesvio(float a,float b,float c,float u){
  float d = 0;
  d += (a-u)*(a-u);
  d += (b-u)*(b-u);
  d += (c-u)*(c-u);
  return sqrt(d / 3);
}

#include <iostream>
#include <iomanip>
using namespace std;

void Troca(float *a, float *b){
  float temp = *a;
  *a = *b;
  *b = temp;
}

void OrdenaTres(float *x,float *y,float *z){
  if(*x > *y) Troca(x,y);
  if(*x > *z) Troca(x,z);
  if(*y > *z) Troca(y,z);
}

int main(){
  float x,y,z;
  cin >> x >> y >> z;
  OrdenaTres(&x,&y,&z);
  cout << fixed << setprecision(1);
  cout << "x = " << x << ", y = " << y << " e z = " << z;
}

#include <cmath>
#include <chrono>

using namespace chrono_literals;

enum Res { RESULTADO_OK = 0, RESULTADO_FALHO = 1 };

Res velocity(int &v,int &w,int l,int &v1,int &v2){
  const double MAX = 100.0;
  if(abs(v) > MAX || abs(w) > MAX) return RESULTADO_FALHO;
  v1 = v + l*w;
  v2 = v - v1;
  return RESULTADO_OK;
}

  cin >> v >> w;
  Res r = velocity(v,w,l,v1,v2);
  if(r == RESULTADO_OK) cout << v1 << " " << v2;
  this_thread::sleep_for(1000ms);
}

bool isPalindrome_v2(char s[]){
  int size=0, left=0;
  while(s[size] != '\0') size++;
  int right = size - 1;

  while(left < right){
    if(s[left] != s[right]) return false;
    left++;
    right--;
  }
  return true;
}

void caps_lock_troca_thr(char s[]){
  for(int i=0; s[i] != '\0'; i++){
    if(s[i] >= 'A' && s[i] <= 'Z') s[i] += ('a'-'A');
    else if(s[i] >= 'a' && s[i] <= 'z') s[i] -= ('a'-'A');
  }
}

void fib_v(long long *vetor, int n){
  long long f, f_1 = 1, f_2 = 0;

  for(int i=0;i<n;i++){
    if(i == 0){ vetor[i] = 0; continue; }
    if(i == 1){ vetor[i] = 1; continue; }
    f = f_1 + f_2;
    f_2 = f_1;
    f_1 = f;
    vetor[i] = f;
  }
}

bool valida(string s){
  if(s.size() != 3) return false;
  for(int i=0;i<3;i++)
    if(s[i] != 'r' && s[i] != 'w' && s[i] != 'x' && s[i] != '-')
      return false;
  return true;
}

int rwx(string s){
  int mask = 0;
  if(s[0] == 'r') mask |= (1 << 2); // 4
  if(s[1] == 'w') mask |= (1 << 1); // 2
  if(s[2] == 'x') mask |= (1 << 0); // 1
  return mask;
}

mask |= 4;     // liga r
mask &= ~2;    // desliga w
mask ^= 1;     // alterna x
